cmake_minimum_required(VERSION 3.16)
project(tpms_libfive LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT EXISTS "${LIBFIVE_SOURCE_DIR}/libfive/src/CMakeLists.txt")
    message(FATAL_ERROR "Set LIBFIVE_SOURCE_DIR to the pinned upstream checkout")
endif()
find_path(EIGEN_INCLUDE_DIRS Eigen/Core HINTS "${TPMS_EIGEN_DIR}" PATH_SUFFIXES eigen3 REQUIRED)
find_path(TPMS_BOOST_INCLUDE_DIR boost/numeric/interval.hpp REQUIRED)
find_package(PNG REQUIRED)
set(Boost_INCLUDE_DIRS "${TPMS_BOOST_INCLUDE_DIR}")
if(MINGW)
    add_compile_definitions(__USE_ISOC99)
    add_compile_options(-Wa,-mbig-obj)
elseif(MSVC)
    add_compile_definitions(_USE_MATH_DEFINES _SCL_SECURE_NO_WARNINGS)
    add_compile_options(/EHsc /wd4267 /wd4244 /wd4305)
endif()
# Build the official kernel unchanged, without Studio or language bindings.
add_subdirectory("${LIBFIVE_SOURCE_DIR}/libfive/src" upstream-kernel)
